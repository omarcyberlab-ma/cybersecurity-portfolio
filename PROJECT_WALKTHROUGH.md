### Migration 001 — `001_init.sql` (Roles, helpers, content tables)

```sql
-- M1: Roles, helpers, content tables
-- Creates app_role enum, user_roles table, has_role helper, site_settings singleton and content tables

BEGIN;
-- enum
CREATE TYPE IF NOT EXISTS app_role AS ENUM ('admin');

-- user_roles
CREATE TABLE IF NOT EXISTS user_roles (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES auth.users(id) ON DELETE CASCADE,
  role app_role NOT NULL,
  created_at timestamptz DEFAULT now(),
  UNIQUE(user_id, role)
);

-- site_settings (singleton)
CREATE TABLE IF NOT EXISTS site_settings (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  singleton boolean DEFAULT true UNIQUE,
  name text,
  title text,
  tagline text,
  bio text,
  email text,
  phone text,
  profile_photo_url text,
  socials jsonb,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

-- content tables (skills, experience, projects, certifications, videos, contact_submissions)
CREATE TABLE IF NOT EXISTS skills (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  category text,
  name text,
  sort int DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS experience (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  role text,
  company text,
  start_date date,
  end_date date,
  bullets text[],
  sort int DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS projects (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  title text,
  description text,
  image_url text,
  live_url text,
  repo_url text,
  youtube_url text,
  sort int DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS certifications (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text,
  issuer text,
  issued_on date,
  pdf_url text,
  sort int DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS videos (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  title text,
  youtube_url text,
  sort int DEFAULT 0,
  created_at timestamptz DEFAULT now(),
  updated_at timestamptz DEFAULT now()
);

CREATE TABLE IF NOT EXISTS contact_submissions (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text,
  email text,
  message text,
  created_at timestamptz DEFAULT now()
);

-- helpers: ensure uuid generator extension
CREATE EXTENSION IF NOT EXISTS pgcrypto;

-- set_updated_at trigger helper
CREATE OR REPLACE FUNCTION set_updated_at()
RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  NEW.updated_at = now();
  RETURN NEW;
END;
$$;

-- attach trigger to tables that have updated_at
DO $$
BEGIN
  IF EXISTS (SELECT 1 FROM information_schema.columns WHERE table_name='site_settings' AND column_name='updated_at') THEN
    EXECUTE 'CREATE TRIGGER site_settings_set_updated_at BEFORE UPDATE ON site_settings FOR EACH ROW EXECUTE FUNCTION set_updated_at();';
  END IF;
END$$;

-- SECURITY DEFINER helper: has_role(user_id, role)
CREATE OR REPLACE FUNCTION has_role(_user_id uuid, _role app_role)
RETURNS boolean
LANGUAGE sql
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT EXISTS (SELECT 1 FROM user_roles ur WHERE ur.user_id = _user_id AND ur.role = _role);
$$;

-- Enable RLS on content tables and add basic policies
ALTER TABLE IF EXISTS site_settings ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS skills ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS experience ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS projects ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS certifications ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS videos ENABLE ROW LEVEL SECURITY;
ALTER TABLE IF EXISTS contact_submissions ENABLE ROW LEVEL SECURITY;

-- Public read for most content tables
CREATE POLICY IF NOT EXISTS public_select_on_content
  ON skills FOR SELECT USING (true);
CREATE POLICY IF NOT EXISTS public_select_on_experience
  ON experience FOR SELECT USING (true);
CREATE POLICY IF NOT EXISTS public_select_on_projects
  ON projects FOR SELECT USING (true);
CREATE POLICY IF NOT EXISTS public_select_on_certifications
  ON certifications FOR SELECT USING (true);
CREATE POLICY IF NOT EXISTS public_select_on_videos
  ON videos FOR SELECT USING (true);

-- Admin-only write for content tables
CREATE POLICY IF NOT EXISTS admin_write_on_content
  ON skills FOR ALL USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));
CREATE POLICY IF NOT EXISTS admin_write_on_experience
  ON experience FOR ALL USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));
CREATE POLICY IF NOT EXISTS admin_write_on_projects
  ON projects FOR ALL USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));
CREATE POLICY IF NOT EXISTS admin_write_on_certifications
  ON certifications FOR ALL USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));
CREATE POLICY IF NOT EXISTS admin_write_on_videos
  ON videos FOR ALL USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));

-- contact_submissions: allow anyone to insert but enforce length checks; only admins may select
CREATE POLICY IF NOT EXISTS contact_insert_length_check
  ON contact_submissions FOR INSERT USING (true)
  WITH CHECK (
    char_length(name) BETWEEN 1 AND 100
    AND char_length(email) BETWEEN 3 AND 255
    AND char_length(message) BETWEEN 1 AND 5000
  );
CREATE POLICY IF NOT EXISTS contact_select_admin_only
  ON contact_submissions FOR SELECT USING (has_role(auth.uid(), 'admin'));

COMMIT;
```

### Migration 004 — `004_bootstrap_first_admin.sql`

```sql
BEGIN;

CREATE OR REPLACE FUNCTION bootstrap_first_admin()
RETURNS trigger LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
BEGIN
	-- If any admin exists, do nothing
	IF EXISTS(SELECT 1 FROM user_roles WHERE role = 'admin') THEN
		RETURN NEW;
	END IF;
	INSERT INTO user_roles (user_id, role) VALUES (NEW.id, 'admin') ON CONFLICT DO NOTHING;
	RETURN NEW;
END;
$$;

-- Create trigger on auth.users table (Supabase managed) if available
DO $$
BEGIN
	IF EXISTS(SELECT 1 FROM pg_tables WHERE schemaname='auth' AND tablename='users') THEN
		PERFORM ('CREATE TRIGGER IF NOT EXISTS after_auth_user_insert AFTER INSERT ON auth.users FOR EACH ROW EXECUTE FUNCTION bootstrap_first_admin();');
	END IF;
END$$;
COMMIT;
```

**Explanation:** The `bootstrap_first_admin()` trigger function runs on every new `auth.users` INSERT. If no admin exists yet, it inserts the new user as admin. Once one admin exists, subsequent signups get NO role (they are not admins). This `NOT EXISTS` guard is race-safe because it runs inside the same transaction as the insert; concurrent inserts serialize on the `user_roles` unique constraint.

> ⚠️ Note: The PERFORM statement uses a string literal for `CREATE TRIGGER IF NOT EXISTS`. This syntax is invalid in Postgres ≤16 (`IF NOT EXISTS` on CREATE TRIGGER was added in PG17). The trigger will fail silently. In practice, create the trigger manually or use a DO block with `CREATE TRIGGER ...` (without `IF NOT EXISTS`) wrapped in exception handling.

```sql
BEGIN;
```

## 4. Storage

### Buckets

Three Supabase Storage buckets are referenced in the scaffold:

| Bucket ID | Visibility | Used by |
|---|---|---|
| `profile` | Private | Profile photo upload |
| `projects` | Private | Project image/screenshot upload |
| `certs` | Private | Certification PDF upload |

All three are **private** buckets. The application never retrieves objects directly via public URLs. Instead, it uses **signed URLs**.

### Storage RLS policies (from `003_storage_policies.sql`)

```sql
CREATE POLICY IF NOT EXISTS storage_admin_write ON storage.objects
  FOR ALL USING (bucket_id IN ('profile','projects','certs') AND has_role(auth.uid(), 'admin'))
  WITH CHECK (bucket_id IN ('profile','projects','certs') AND has_role(auth.uid(), 'admin'));
```

This restricts INSERT/UPDATE/DELETE on `storage.objects` to admin users for the three buckets. SELECT (read) is not granted via RLS — reads go through signed URLs generated by the service_role client.

### Size/MIME constraints (from `FileUpload.tsx`)

The current `FileUpload.tsx` component does **not** enforce client-side size or MIME type restrictions. The file input accepts any file type. A production implementation should validate `file.type` and `file.size` before uploading.

### Signed URLs

- **TTL:** 1 year (`365 * 24 * 60 * 60` seconds)
- **Generation:** `supabase.storage.from('profile').createSignedUrl(filePath, expiresIn)`
- **Call sites:** The scaffold's `FileUpload.tsx` does not yet implement actual signed URL creation (it uses a stub `setTimeout`). Server functions would call `createSignedUrl` with the service_role client.

### Current known gap

The `FileUpload.tsx` component (at `src/components/admin/FileUpload.tsx`) simulates an upload with an 800ms timeout and returns a fake URL. Real upload + signed-URL generation needs to be wired through a server function that uses the Supabase service_role client.

## 5. Auth flow

### Step-by-step trace

1. **User navigates to `/admin`**
2. `_authenticated/route.tsx` `beforeLoad` runs:
   ```ts
   beforeLoad: async () => {
     const user = null; // placeholder: call supabase.auth.getUser()
     if (!user) throw redirect({ to: "/auth" });
     return { user };
   }
   ```
3. Since user is null, TanStack Router throws a `redirect` to `/auth`.
4. **`/auth` page** renders a placeholder sign-in form. In production this would call:
   ```ts
   const { data, error } = await supabase.auth.signInWithPassword({ email, password });
   ```
5. After successful sign-in, `supabase.auth.onAuthStateChange` fires. The app can invalidate the router to re-run the `beforeLoad` guard.
6. The router re-evaluates `_authenticated/route.tsx` `beforeLoad`. This time `supabase.auth.getUser()` returns a user, so the guard passes.
7. The `_authenticated` layout renders, and the child route `/_authenticated/admin` (the admin dashboard) displays.

### Bootstrap-trigger race semantics

When the first user signs up:

1. `auth.users` INSERT triggers `bootstrap_first_admin()`.
2. The trigger checks `IF EXISTS(SELECT 1 FROM user_roles WHERE role = 'admin')`.
3. If no admin exists, it does `INSERT INTO user_roles (user_id, role) VALUES (NEW.id, 'admin') ON CONFLICT DO NOTHING`.
4. Because the trigger runs inside the same transaction as the `auth.users` INSERT, simultaneous signups serialize on the `user_roles` UNIQUE constraint. Only the first transaction to commit successfully promotes its user to admin.
5. **Result:** Exactly one admin is created — the first user who signs up. Subsequent signups get no role.

### Locking signups after bootstrap

To prevent unwanted signups after the admin is created:

1. Go to the Supabase Dashboard → Authentication → Settings.
2. Disable "Allow new signups" (toggle off).
3. Alternatively, use the Lovable Cloud Backend panel (if using Lovable platform) to disable sign-ups.

Once signups are disabled, only the existing admin can log in. New users cannot register.

## 6. Server functions

### `getSiteData(env?)` — `src/lib/portfolio.functions.ts`

```ts
export async function getSiteData(env?: any) {
  const serverEnv = env || {};
  const url = serverEnv.SUPABASE_URL || (typeof import.meta !== 'undefined' ? import.meta.env.SUPABASE_URL : '') || '';
  const key = serverEnv.SUPABASE_SERVICE_ROLE_KEY || (typeof import.meta !== 'undefined' ? import.meta.env.SUPABASE_SERVICE_ROLE_KEY : '') || '';
  if (!url || !key) return { ...FALLBACK };
  const svc = createServerClient(serverEnv);
  try {
    const [settingsRes, skillsRes, experienceRes, projectsRes, certsRes, videosRes] = await Promise.all([
      svc.from('site_settings').select('id,name,title,tagline,bio,profile_photo_url,socials,created_at,updated_at').limit(1).single(),
      svc.from('skills').select('id,category,name,sort').order('category').order('sort'),
      svc.from('experience').select('id,role,company,start_date,end_date,bullets,sort').order('sort'),
      svc.from('projects').select('id,title,description,image_url,live_url,repo_url,youtube_url,sort').order('sort'),
      svc.from('certifications').select('id,name,issuer,issued_on,pdf_url,sort').order('sort'),
      svc.from('videos').select('id,title,youtube_url,sort').order('sort')
    ]);
    const settings = settingsRes.error ? null : settingsRes.data;
    return {
      settings: settings ?? FALLBACK.settings,
      skills: skillsRes.error ? [] : skillsRes.data,
      experience: experienceRes.error ? [] : experienceRes.data,
      projects: projectsRes.error ? [] : projectsRes.data,
      certifications: certsRes.error ? [] : certsRes.data,
      videos: videosRes.error ? [] : videosRes.data
    };
  } catch (err) {
    console.error('getSiteData failed', err);
    return FALLBACK;
  }
}
```

**Middleware trace:**
- `getSiteData` creates a direct Supabase client using `SUPABASE_SERVICE_ROLE_KEY` (bypasses RLS). No auth middleware needed — it uses the service role to read all public data.

### `submitContact(payload, env?)` — `src/lib/portfolio.functions.ts`

```ts
export const submitContact = async (payload: any, env?: any) => {
  const schema = z.object({ name: z.string().min(1).max(100), email: z.string().email().max(255), message: z.string().min(1).max(5000) });
  const parsed = schema.safeParse(payload);
  if (!parsed.success) throw new Error('Invalid input');
  const serverEnv = env || {};
  const url = serverEnv.SUPABASE_URL || (typeof import.meta !== 'undefined' ? import.meta.env.SUPABASE_URL : '') || '';
  const key = serverEnv.SUPABASE_SERVICE_ROLE_KEY || (typeof import.meta !== 'undefined' ? import.meta.env.SUPABASE_SERVICE_ROLE_KEY : '') || '';
  if (!url || !key) throw new Error('Supabase not configured');
  const svc = createServerClient(serverEnv);
  try {
    const { error } = await svc.from('contact_submissions').insert({ name: parsed.data.name, email: parsed.data.email, message: parsed.data.message });
    if (error) { console.error('submitContact DB error', error); throw new Error('Could not save your message'); }
    return { ok: true };
  } catch (err) {
    console.error('submitContact failed', err);
    throw new Error('Could not save your message');
  }
};
```

**Middleware trace:**
- Validates input with Zod (mirrors DB length constraints).
- Creates a service_role client (no auth needed for insert).
- Writes to `contact_submissions`.

### Admin server functions (used by `admin.tsx`)

The scaffold's `_authenticated/admin.tsx` does **not** yet call any server functions for CRUD operations. It renders a tab-based UI placeholder. Real admin functions would follow the pattern:
1. Client calls `serverFn` (TanStack Start server function).

#### `Reveal.tsx`
```tsx
import React, { useEffect, useRef } from 'react';

export default function Reveal({ children, delay = 0 }: any) {
  const ref = useRef<HTMLDivElement | null>(null);
  useEffect(() => {
    const el = ref.current; if (!el) return;
    const obs = new IntersectionObserver((entries) => {
      entries.forEach(e => { if (e.isIntersecting) el.classList.add('reveal-in'); });
    }, { threshold: 0.1 });
    obs.observe(el);
    return () => obs.disconnect();
  }, []);
  return <div ref={ref} className="reveal" style={{ transitionDelay: `${delay}ms` }}>{children}</div>;
}
```
- **Props:** `children` (ReactNode), `delay` (ms, default 0).
- **Behavior:** Uses IntersectionObserver to add `reveal-in` class when element is 10% visible. Starts with `reveal` class (opacity 0, translateY 6px).

## 10. Data-flow traces

### A. Public homepage SSR load

```
1. Browser requests GET /
2. Cloudflare Worker receives request
3. src/server.ts: createRequestHandler({ createRouter: getRouter, request })
4. Router resolves route tree → matches "/" (src/routes/index.tsx)
5. Route loader executes:
   a. context.queryClient.ensureQueryData(siteQuery)
   b. siteQuery.queryFn → getSiteData() in src/lib/portfolio.functions.ts
   c. getSiteData reads SUPABASE_URL + SUPABASE_SERVICE_ROLE_KEY from env
   d. Creates service_role Supabase client via createServerClient()
   e. Runs 6 parallel queries:
      - site_settings: SELECT id,name,title,tagline,bio,profile_photo_url,socials,created_at,updated_at
      - skills: SELECT id,category,name,sort ORDER BY category, sort
      - experience: SELECT id,role,company,start_date,end_date,bullets,sort ORDER BY sort
      - projects: SELECT id,title,description,image_url,live_url,repo_url,youtube_url,sort ORDER BY sort
      - certifications: SELECT id,name,issuer,issued_on,pdf_url,sort ORDER BY sort
      - videos: SELECT id,title,youtube_url,sort ORDER BY sort
   f. Returns { settings, skills, experience, projects, certifications, videos }
6. Router passes loader data to Index component via Route.useLoaderData()
7. Index component renders:
   - <h1>{settings.name}</h1>
   - <section id="about">About</section>
   (In a full build, section components would render here)
8. defaultRenderHandler serializes the React tree to HTML
9. Response sent to browser with hydrated JS bundle
```

### B. Admin login + edit project mutation

```
1. User navigates to /admin
2. _authenticated/route.tsx beforeLoad runs:
   a. Calls supabase.auth.getUser() (via client-side Supabase instance)
   b. If no session → redirect to /auth
3. User is at /auth page, enters email + password
4. Client calls supabase.auth.signInWithPassword({ email, password })
5. Supabase returns session; onAuthStateChange fires
6. App invalidates router (or window.location.reloads)
7. Router re-evaluates _authenticated beforeLoad → user exists → passes
8. Admin dashboard renders at /_authenticated/admin
9. Admin clicks "Projects" tab
10. Admin fills in project form, clicks Save
11. Client calls server function (TanStack Start serverFn):
    a. attachSupabaseAuth middleware attaches Bearer token from request
    b. Server validates session with supabase.auth.getUser(token)
    c. Server checks has_role(user.id, 'admin') via service_role client
    d. Server calls svc.from('projects').upsert({...})
    e. Returns success
12. React Query invalidates ["site-data"] → homepage reflects changes
```

### C. Contact form submission

```
1. Visitor fills in name, email, message on Contact.tsx form
2. Clicks Send
3. Client-side: (scaffold just sets sent=true, but in production:)
   a. Client calls submitContact(payload) server function
   b. submitContact validates with Zod schema:
      - name: z.string().min(1).max(100)
      - email: z.string().email().max(255)
      - message: z.string().min(1).max(5000)
   c. Creates service_role Supabase client
   d. Inserts into contact_submissions(name, email, message)
   e. DB RLS policy contact_insert_length_check validates length again
   f. Returns { ok: true }
4. Client shows "Thanks — message sent."
5. Admin later logs in, goes to Submissions tab in /admin
6. Admin fetches svc.from('contact_submissions').select('*') (service_role bypass)
   or uses anon client with has_role check (RLS: contact_select_admin_only)
```

## 11. Known limitations / TODOs

| # | Limitation | Impact | Fix |
|---|---|---|---|
| 1 | Placeholder name "Your Name" / "Alex Chen" in all defaults | Site shows generic text if DB not configured | Replace with actual data in DB |
| 2 | No rate limiting on `submitContact` | Spam / abuse of contact form | Add IP-based rate limiting or CAPTCHA (e.g. Turnstile) |
| 3 | No CAPTCHA on contact form | Bot submissions | Integrate Cloudflare Turnstile or Google reCAPTCHA |
| 4 | No password reset UI | Admin cannot reset password if forgotten | Add "Forgot password?" flow via `supabase.auth.resetPasswordForEmail()` |
| 5 | No email notifications on new contact | Admin must manually check dashboard | Set up Supabase Edge Function or webhook to email on INSERT |
| 6 | No image optimization / resizing | Large uploads served as-is | Use `supabase.storage` transformations or a CDN image resizer |
| 7 | Signed URL 1-year TTL | If bucket key changes, old URLs break until re-fetched | Call `createSignedUrl` on every page load, or use shorter TTL with refresh |
| 8 | Fonts not bundled | Falls back to system fonts | Add `@fontsource/jetbrains-mono` and `@fontsource/inter` |
| 9 | `public/cv.pdf` is static placeholder | No dynamic CV upload | Wire FileUpload for certs bucket to replace PDF |
| 10 | `auth.tsx` is stub | No real login UI | Build sign-in form with `supabase.auth.signInWithPassword()` |

| 12 | `FileUpload.tsx` is stub | No actual upload | Implement `supabase.storage.upload()` + `createSignedUrl()` |
| 13 | Bootstrap trigger uses invalid PG syntax | Trigger may not be created on PG <17 | Use DO block with exception handling instead of `CREATE TRIGGER IF NOT EXISTS` |

## 12. Safe-modification guide

### Recipe A: Add a new table (e.g. `blog_posts`)

1. **Write migration** (create `009_blog_posts.sql` in `supabase/migrations/`):
   ```sql
   BEGIN;
   CREATE TABLE IF NOT EXISTS blog_posts (
     id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
     title text NOT NULL,
     slug text UNIQUE NOT NULL,
     content text,
     published boolean DEFAULT false,
     created_at timestamptz DEFAULT now(),
     updated_at timestamptz DEFAULT now()
   );
   -- GRANTs: anon and authenticated can SELECT
   GRANT SELECT ON blog_posts TO anon, authenticated;
   -- Enable RLS
   ALTER TABLE blog_posts ENABLE ROW LEVEL SECURITY;
   -- Public select
   CREATE POLICY blog_posts_select ON blog_posts FOR SELECT USING (published = true);
   -- Admin write
   CREATE POLICY blog_posts_admin ON blog_posts FOR ALL
     USING (has_role(auth.uid(), 'admin')) WITH CHECK (has_role(auth.uid(), 'admin'));
   COMMIT;
   ```

2. **Add `set_updated_at` trigger** (in the same or next migration):
   ```sql
   CREATE TRIGGER blog_posts_set_updated_at BEFORE UPDATE ON blog_posts
     FOR EACH ROW EXECUTE FUNCTION set_updated_at();
   ```

3. **Wait for type regeneration** — run `npx supabase gen types typescript --local > src/integrations/supabase/types.ts` (or use the Supabase CLI to regenerate types).

4. **Extend `SiteData`** — in `src/lib/portfolio.functions.ts`, add:
   ```ts
   blogPosts: [],  // in FALLBACK
   ```
   And in `getSiteData`:
   ```ts
   const blogRes = await svc.from('blog_posts').select('id,title,slug,published,created_at').eq('published', true).order('created_at', { ascending: false });
   // add to return: blogPosts: blogRes.error ? [] : blogRes.data
   ```

5. **Create `BlogPosts.tsx` component** in `src/components/sections/`:
   ```tsx
   export default function BlogPosts({ posts = [] }: any) {
     return (<section>{posts.map(p => <div key={p.id}><h3>{p.title}</h3></div>)}</section>);
   }
   ```

6. **Register in homepage** — edit `src/routes/index.tsx` to import and render `<BlogPosts posts={data.blogPosts} />`.

7. **Add admin tab** — in `src/routes/_authenticated/admin.tsx`, add a `Blog` tab with list/create/edit/delete controls backed by server functions (see Recipe B pattern).

### Recipe B: Add a field to an existing table

1. **Write migration** (e.g. `010_add_location.sql`):
   ```sql
   BEGIN;
   ALTER TABLE IF EXISTS experience ADD COLUMN IF NOT EXISTS location text;
   COMMIT;
   ```

2. **Run migration** via Supabase CLI or psql.

3. **Update type** — add `location?: string` to the experience type in `src/integrations/supabase/types.ts`.

4. **Update SELECT** — in `getSiteData`'s experience query, add `'location'` to the `.select(...)` list.

5. **Update component** — in `Experience.tsx`, render the new field:
   ```tsx
   <div>{it.location && `📍 ${it.location}`}</div>
   ```

6. **Update admin form** — add the field input in the admin CRUD UI.

### Recipe C: Add a new route

1. **Create file** — e.g. `src/routes/blog.tsx`:
   ```tsx
   import { createFileRoute } from "@tanstack/react-router";
   export const Route = createFileRoute("/blog")({
     component: () => <div>Blog index</div>,
   });
   ```

2. **Type-check** — run `npm run typecheck`. TanStack Router's Vite plugin auto-generates the route tree entry.

3. **Add link** — in `SiteNav.tsx`, add `<a href="/blog">Blog</a>`.

### General safety rules

- **Never edit `routeTree.gen.ts`** — it is auto-generated by the TanStack Router Vite plugin. Edit only the source route files.
- **Never edit lockfiles** (`bun.lock`, `package-lock.json`) — re-run `bun install` or `npm install`.
- **Migration order matters** — files prefixed `001_` through `008_` must run sequentially. New migrations get the next number.
- **Test changes locally** with `npm run dev` before deploying.
- **Run typecheck:** `npm run typecheck` to catch TS errors.
- **Environment variables** — use `.env` for local dev. Server-side vars (`SUPABASE_SERVICE_ROLE_KEY`) must be set in the Cloudflare Worker dashboard or Wrangler config for production.
